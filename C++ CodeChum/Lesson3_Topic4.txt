    cin>>size;

    cout<<"Enter the elements of the array:"<<endl;
    
    for(int i=0;i<size;i++){
        int z=i+1;
        cout<<"Enter element #"<<z<<": ";
        cin>>x[i];
    }
    cout<<"Enter a value to search for: ";
    cin>>search;
    for(int i=0;i<size;i++){
        if(x[i]==search){
            found=true;
        }
    }
    if(found==true){
        cout<<"Found!";
    }
    else{
        cout<<"Sorry, "<<search<<" not found.";
    }
    
}

#include<iostream>
#include<iomanip>
using namespace std;

int main(){

    int x[20];
    int size=0;
    double temp=0;
    float average=0;

    cout<<"Enter the size of the array: ";
    cin>>size;
    
    for(int i=0;i<size;i++){
        int z=i+1;
        cout<<"Enter element #"<<z<<": ";
        cin>>x[i];
    }
    for(int i=0;i<size;i++){
        temp+=x[i];
        average=temp/size;
    }
    cout<<"Sum: "<<temp<<endl;
    cout<<fixed<<setprecision(2)<<"Average: "<<average;
    
}

#include<iostream>
#include<iomanip>
using namespace std;

int main(){
    int max = 20;
    int x[max];
    int size=0;

    
    cout<<"Enter the size of the array: ";
    cin>>size;
    if(size<=0||size>max){
        return 1;
    }
    
    for(int i=0;i<size;i++){
        int z=i+1;
        cout<<"Enter element #"<<z<<": ";
        cin>>x[i];
    }
    cout<<"Even: ";
    for(int i=0;i<size;i++){
        if(x[i]%2==0){
            cout<<x[i];
            if(i<size-2){
                cout<<", ";
            }
        }
    }
    cout<<endl;
    cout<<"Odd: ";
    for(int i=0;i<size;++i){
        if(x[i]%2!=0){
            cout<<x[i];
            if(i<size-2){
                cout<<", ";
            }
        }
    }
    
}
