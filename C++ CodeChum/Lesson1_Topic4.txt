#include<iostream>
#include<iomanip>
using namespace std;

int main(){

    double a,b;
    char op;

    cout<<"Enter first number: ";
    cin>>a;
    cout<<"Enter second number: ";
    cin>>b;
    cout<<"Enter operation (+, -, *, or /): ";
    cin>>op;

    switch(op){
        case '+':
            cout<<fixed<<setprecision(2)<<"Result: "<<a+b;
            break;
        case '-':
            cout<<fixed<<setprecision(2)<<"Result: "<<a-b;
            break;
        case '*':
            cout<<fixed<<setprecision(2)<<"Result: "<<a*b;
            break;
        case '/':
            cout<<fixed<<setprecision(2)<<"Result: "<<a/b;
            break;
        default:
            cout<<"Result: -999.99";
            break;
        
    }
}
