#include<iostream>
using namespace std;

void reverseInt(int a){
    int reverse=0;
    int remainder;
    while(a!=0){
        remainder=a%10;
        reverse = reverse*10+remainder;
        a/=10;
    }
    cout<<reverse;
}
int main(){
    int x;
    cout<<"Enter an integer: ";
    cin>>x;
    reverseInt(x);
}
