    string x;
    int y;

    cout<<"Enter an integer: ";
    cin>>x;

    for(char c : x){

        if(isdigit(c)){
            if(c%3==0){
                continue;
                sum+=(c -'0');
            }
            else{
                sum+=(c -'0');
            }
        }
    }
    cout<<"Sum of all digits: "<<sum;
