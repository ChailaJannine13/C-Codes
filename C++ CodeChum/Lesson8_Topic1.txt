#include <iostream>
using namespace std;

enum Week {
    MON = 0,    // 0
    TUE = 1,    // 1 
    WED = 2,    // 2
    THU = 3,    // 3
    FRI = 4,    // 4
    SAT = 5,    // 5
    SUN = 6    // 6
};

int main() {
	int day;
    cout << "Enter day number: ";
    cin >> day;

    switch(day){
        case MON:
            cout<<"Monday";
            break;
        case TUE:
            cout<<"Tuesday";
            break;
        case WED:
            cout<<"Wednesday";
            break;
        case THU:
            cout<<"Thursday";
            break;
        case FRI:
            cout<<"Friday";
            break;
        case SAT:
            cout<<"Saturday";
            break;
        case SUN:
            cout<<"Sunday";
            break;
        default:
            cout<<"Not valid";
            break;
    }


    return 0;
    
    

}
