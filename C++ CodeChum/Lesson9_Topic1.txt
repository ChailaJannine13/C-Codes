#include <iostream>
#include <fstream>
#include <cctype>
#include <string>
#include <cstdio>
#include <sys/stat.h>

using namespace std;

void createdir(const string& dir){
    struct stat info;
    if(stat(dir.c_str(), &info) !=0){
        #if defined(_WIN32)
            _mkdir(dir.c_str());
        #else
            mkdir(dir.c_str(),0777);
        #endif
    }
    else if(!(info.st_mode & S_IFDIR)){
        cerr<<"Error: "<<dir<<" is not a directory!"<<endl;
        exit(1);
    }
    
}

#include <iostream>
#include <fstream>
#include <unordered_set>
#include <vector>
#include <string>

using namespace std;

int main() {
    int n;
    string name;
    unordered_set<string> nameSet;  // For checking duplicates
    vector<string> names;  // For maintaining the order of input

    // Ask the user for the number of names
    cout << "Enter the number of names: ";
    cin >> n;
    cin.ignore(); // Ignore newline character left in the input buffer

    // Get the names from the user
    for (int i = 0; i < n; ++i) {
        cout << "Enter name " << i + 1 << ": ";
        getline(cin, name);
        // Check if the name is not a duplicate
        if (nameSet.find(name) == nameSet.end()) {
            names.push_back(name);  // Add name to vector
            nameSet.insert(name);  // Add name to set to keep track of duplicates
        }
    }

    // Open a file to write the names
    ofstream file("people.txt");
    if (file.is_open()) {
        for (const auto& unique_name : names) {
            file << unique_name << endl; // Write each unique name to the file in the order entered
        }
        file.close();
    } else {
        cerr << "Unable to open file" << endl;
    }

    return 0;
}

#include <fstream>
#include <iostream>
#include <vector>
#include <cmath>

using namespace std;

int main(void) {

    ofstream dataFile;
    dataFile.open("data.txt");
    dataFile << "5\n3\n2\n";
    dataFile.close();

    ifstream inFile("data.txt");
    vector<int> numbers;
    int number;
    while (inFile >> number) {
        numbers.push_back(number);
    }
    inFile.close();


    vector<int> squares;
    for (int num : numbers) {
        squares.push_back(pow(num, 2));
    }

    ofstream outFile("data.txt", ios::app);
    outFile << "\n"; 
    for (int square : squares) {
        outFile << square << '\n';
    }
    outFile.close();

    return 0;
}
