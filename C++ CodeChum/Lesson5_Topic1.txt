int min(int arr[], int n, int minval){
    minval = arr[0];
    for(int i=1;i<n;++i){
        if(arr[i]<minval){
            minval=arr[i];
        }
    }
    return minval;
}

#include<iostream>

int sumatrange(int arr[],int size, int start, int end, int sum);
int main(){
    int arrr[10];
    int sizee;
    int starting;
    int ending;
    int sum;

    cout<<"Enter size of array: ";
    cin>>sizee;
    cout<<"Enter array values: ";
    for(int i=0;i<sizee;i++){
        cin>>arrr[i];
    }
    cout<<"Enter starting range: ";
    cin>>starting;
    cout<<"Enter end range: ";
    cin>>ending;
    sumatrange(arrr,sizee,starting,ending,sum);
    

}

int sumatrange(int arr[],int size, int start, int end, int sum){

    for(int i=start-1;i<size;i++){
        sum+=arr[i];
        if(end==i){
            break;
        }
    }
    cout<<"Sum at range: "<<sum;
    return sum;
}
