#include <iostream>
#include <fstream>
#include <sstream>

using namespace std;

int main() {
    ofstream csvFile("numbers.csv");
    int count = 0;
    for (int i = 1; i <= 100; i++) {
        csvFile << i;
        count++;
        if (count % 10 == 0 && i != 100) {
            csvFile << endl;
        } else if (i != 100) {
            csvFile << ",";
        }
    }
    csvFile.close();

    // Reading from file
    ifstream readFile("numbers.csv");
    string line;
    int rowCount = 0;
    while (getline(readFile, line)) {
        stringstream ss(line);
        string number_str;
        int numbersPrinted = 0;
        while (getline(ss, number_str, ',')) {
            int number = stoi(number_str); // Convert string to integer
            if (number % 2 == 0) { // Check if the number is even
                cout << number;
                numbersPrinted++;
                if (numbersPrinted < 5) { // Print comma if not the last number in the row
                    cout << ", ";
                } else { // Print new line if it's the last number in the row
                    cout << endl;
                }
            }
        }
    }

    readFile.close();

    return 0;
}
