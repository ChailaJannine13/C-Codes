#include<iostream>
#include<vector>
#include<sstream>
#include<iomanip>
#include<limits>
using namespace std;
std::pair<int, double> average(const std:: vector<int>&numbers){
    int sum = 0;
    for(int num:numbers){
        sum+=num;
    }
    double average = static_cast<double>(sum)/numbers.size();
    return{sum, average};
}
int main(){
    int x;
    
    std::cout<<"Enter size of array: ";
    std::cin>>x;

    std::cin.ignore(std::numeric_limits<std::streamsize>::max(),'\n');
    std::cout<<"Enter the numbers: ";
    std::string input;
    std::getline(std::cin,input);
    std::istringstream iss(input);

    int num;
    std::vector<int>numbers;
    while(iss>>num){
        numbers.push_back(num);
    }
    auto result = average(numbers);

    std::cout<<fixed<<setprecision(2)<<"Average: "<<result.second;
    return 0;
}
