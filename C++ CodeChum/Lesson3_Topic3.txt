#include <iostream>
#include <iomanip>
using namespace std;

int main(){
    int c,r;
    int x[10][10];
    int XL = x[0][0];
    cout<<"Enter number of rows: ";
    cin>>r;
    cout<<"Enter number of columns: ";
    cin>>c;

    
    for(int y=0; y<r;y++){
           for(int z=0;z<c;z++){
            cout<<"Enter element at row "<<y<<", column "<<z<<": ";
            cin>>x[y][z];
        }
    }
     for(int y=0;y<r;y++){
         for(int z=0;z<c;z++){
            if(x[y][z]>XL){
                XL=x[y][z];
            }
         }    
     }
    cout<<"Largest: "<<XL;
    return 0;
}
