#include<iostream>
using namespace std;

int factorial(int i){
    if(i==0||i==1){
        return 1;
    }
    else{
        return i*factorial(i-1);
    }
    
}

int main(){

    int x,y;
    cout<<"Enter a number: ";
    cin>>x;
    y=x;
    x=factorial(x);
    cout<<"Factorial of "<<y<<": "<<x;
    
    
}

#include<iostream>
using namespace std;

void printRecursively(int current, int n) {
    if(current <= n) {
        cout << current << " ";
        return printRecursively(current +1, n);
    }
}

int main(void) {
    int n;
    
    cout << "Enter n: ";
    cin >> n;
    
    printRecursively(1, n);
    
    return 0;
}

#include <iostream>
using namespace std;

int fib(int n) {
    if (n <= 1) {
        return n;
    }

    return fib(n - 1) + fib(n - 2);
}
 
int main(void) {
    int n;
    
    cout << "Enter which nth term: ";
    cin >> n;
    
    cout << "Nth term = " << fib(n);

    return 0;
}

#include<iostream>
using namespace std;

int computeSum(int n) {
    if (n != 0) {        
        return n + computeSum(n - 1);
    } else {
        return n;
    }
}

int main(void) {
    int n;
    
    cout << "Enter n: ";
    cin >> n;
    
    cout << "Sum = " << computeSum(n);
    
    return 0;
}
