#include <iostream>
using namespace std;

int main(void) {
    while(1) {
        int a, b;
        char op;
        
        cout << "Enter a: ";
        cin >> a;
        
        cout << "Enter b: ";
        cin >> b;
        
        cout << "Enter operator: ";
        cin >> op;
        
        if(op == '+') {
            cout << a << " + " << b << " = " << (a + b) << endl << endl;
        } else if (op == '-') {
            cout << a << " - " << b << " = " << (a - b) << endl << endl;
        } else if (op == '*') {
            cout << a << " * " << b << " = " << (a * b) << endl << endl;
        } else if (op == '/') {
            cout << a << " / " << b << " = " << (a / b) << endl << endl;
        }
        else{
            break;
        }
    }

    return 0;
}
