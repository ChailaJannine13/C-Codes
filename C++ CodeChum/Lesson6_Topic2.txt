#include "date.h"

int compareDates(Date a, Date b) {


    if(a.year<b.year){
        return -1;
         if(a.year>b.year){
           return 1;
         }   
    }
    else if(a.year>b.year){
        return 1;
         if(a.year<b.year){
           return -1;
         }   
    }
    else if(a.month<b.month){
        return -1;
        if(a.month>b.month){
            return 1;
        }
    }
    else if(a.month>b.month){
        return 1;
        if(a.month<b.month){
            return -1;
        }
    }
    else if(a.day<b.day){
        return -1;
        if(a.day>b.day){
        return 1;
        }
    }
    else if(a.day>b.day){
        return 1;
        if(a.day<b.day){
        return -1;
        }
    }
    
    return 0;
}

    if(i==5){
        for(int x = 0; x<i; x++){
            cout<<book.title<<" by "<<book.author<<" ("<<book.numPage<<" pages)"<<endl;
        }
    }
    else if(i==7){
        for(int x = 0; x<i; x++){

        for(int x = 0; x<i; x++){
